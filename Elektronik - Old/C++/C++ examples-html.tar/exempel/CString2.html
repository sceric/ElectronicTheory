//! demonsterar hur C-strängar skapas, pekararitmetik
#include <iostream>

int main () {
  char Hello1[] = "Hello!";
  char * Hello2 = "Hello again!";
  char Hello3[5];

  *(Hello3 + 0) = 'H';
  *(Hello3 + 1) = 'e';
  *(Hello3 + 2) = 'j';
  *(Hello3 + 3) = '!';
  *(Hello3 + 4) = '\0';

  cout << Hello1 << ' ' << Hello2 << ' ' << Hello3 << endl;
}
