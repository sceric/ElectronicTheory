#include <iostream.h>

// prototyper för våra två funktioner
float RaknaArea (float Sida1, float Sida2);
void SkrivSvar (float Area);

int main () {
  // beräkna arean
  float Area = RaknaArea ( 5.5, 10 );

  // skriv ut svaret
  SkrivSvar ( Area );
}

float RaknaArea (float Sida1, float Sida2) {
  // beräkna och returnera en area av en rektangel
  return Sida1 * Sida2;
}

void SkrivSvar (float Area) {
  cout << "Arean är: " << Area << endl;
}
