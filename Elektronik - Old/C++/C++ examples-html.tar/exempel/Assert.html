//! testar makrot 'assert()'
#include <iostream>
#include <assert.h>

void Calculate (int Tal) {
  assert ( Tal > 10 );
  cout << Tal << " är ok" << endl;
}

int main () {
  int Tal;

  // läs in tal i all evighet
  while ( true ) {
	cout << "Ge ett tal > 10: ";
	cin >> Tal;
	Calculate ( Tal );
  }
}
