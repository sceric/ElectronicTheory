#include <iostream>

class Coordinate {
public:
  // konstruktorer
  Coordinate (float X, float Y) { m_X = X; m_Y = Y; };
  Coordinate (const Coordinate & Original) { m_X = Original.x (); m_Y = Original.y (); };
  Coordinate () { m_X = 0; m_Y = 0; };

  // accessera medlemmar
  float x () const { return m_X; };
  float y () const { return m_Y; };
  void setX (float X) { m_X = X; };
  void setY (float Y) { m_Y = Y; };

private:
  // medlemmar
  float m_X;
  float m_Y;
};

// skriv ut en koordinat
void printCoordinate (Coordinate C) {
  cout << "X = " << C.x () << ", Y= " << C.y () << endl;
}


int main () {
  Coordinate C1 ( 1, 2);
  Coordinate C2 ( C1 );
  Coordinate C3 = C1;

  // skriv ut de olika koordinaterna
  printCoordinate ( C1 );
  printCoordinate ( C2 );
  printCoordinate ( C3 );
}
