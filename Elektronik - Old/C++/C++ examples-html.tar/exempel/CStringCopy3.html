//! kopierar en C-sträng med en funktion copy(). Felaktig kopiering!
#include <iostream>

void copy (char * Source, char * Destination) {
  // iterera över och kopiera alla tecken
  for ( unsigned int Index = 0; Index <= strlen ( Source ); Index++ ) {
    // kopiera ett tecken
    Destination[Index] = Source[Index];
  }
}
  
int main () {
  // aj aj, Hello2 är för kort!
  char Hello1[] = "Hej! Detta är en lååång sträng.";
  char Hello2[1];

  // utför kopieringen 
  copy ( Hello1, Hello2 );

  cout << "Hello1 = '" << Hello1 << "', " << endl
	   << "Hello2 = '" << Hello2 << "'" << endl;
}
