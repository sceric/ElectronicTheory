//! beräknar längden på en C-sträng
#include <iostream>
#include <string.h>

int main () {
  char * S1 = "Hej!";
  char * S2 = "Hello world, this is main() calling";

  cout << S1 << " är " << strlen (S1) << " tecken lång." << endl;
  cout << S2 << " är " << strlen (S2) << " tecken lång." << endl;
}
