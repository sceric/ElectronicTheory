//! läser in två tecken från tangentbordet
#include <iostream>

int main () {

  char Tecken1, Tecken2;

  // upprepa slingan ända tills användaren ger in tecknet 'q'
  do {
	// läs in ett tecken
	cout << "Ge två tecken: ";
	cin >> Tecken1 >> Tecken2;
	cout << "Du gav in tecknen '" << Tecken1 << "' och '" << Tecken2 << "'." << endl;
  }
  while ( Tecken1 != 'q' && Tecken2 != 'q' );
}
