//! testar konkatenering av strängar med +=
#include <iostream>
#include <string>

string fulltNamn (const string & Fornamn, const string & Efternamn) {
  string Namn = Fornamn;
  Namn += ' ';
  Namn += Efternamn;

  // ge tillbaka det fullständiga namnet
  return Namn;
}

int main () {
  string Namn = fulltNamn ( "Frodo", "Bagger" );
  string Text = "Fullständigt namn: " + Namn;
  
  cout << Text << endl;
}
