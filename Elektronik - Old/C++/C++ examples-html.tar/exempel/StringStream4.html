//! demonstrerar konvertering till tal med hjälp av istrstream
#include <strstream>
#include <string>

int main () {
  string Text;
  string Rest;
  int Tal;
  
  // läs in en rad
  cout << "Ge ett tal: ";
  cin >> Text;
  
  // skapa stream
  istrstream Stream ( Text.c_str () );

  // läs ett tal om det går
  if ( Stream >> Tal ) {
    // talet är ok
    cout << "Läste talet " << Tal << endl;

    // finns det mera text kvar i buffern?
    if ( Stream >> Rest ) {
      // jep, mera text kvar, så det är kanske inte ett ok tal?
      cout << "Texten '" << Rest << "' blev oläst." << endl;
    }
  }
  else {
    // inget giltigt tal
    cout << "Texten '" << Text << "' är inte ett tal." << endl;
  }
}
