//! läser en given fil tecken för tecken och räknar tecknen
#include <fstream>
#include <stdlib.h>

int main (int argc, char * argv[]) {
  int Antal = 0;
  char Tecken;
  
  // har vi tillräckligt parametrar?
  if ( argc != 2 ) {
    cout << "Fel antal parametrar!" << endl;
    cout << "Användning: " << argv[0] << " filnamn" << endl;
    exit ( EXIT_FAILURE );
  }

  // öppna en fil
  ifstream Data ( argv[1] );
  if ( ! Data ) {
    // kunde inte öppna filen
    cout << "Kunde inte öppna filen: " << argv[1] << endl;
    exit ( EXIT_FAILURE );
  }

  // iterera över all tecken i filen
  while ( Data.get ( Tecken ) ) {
    // öka antalet lästa tecken
    Antal++;
  }

  // skriv ett svar
  cout << "Läste " << Antal << " tecken." << endl;
}
