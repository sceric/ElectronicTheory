//! använder string::find() för att tokenisera en sträng
#include <fstream>
#include <string>
#include <iomanip>

int main () {
  string Rad;
  string::size_type Index1;
  string::size_type Index2;
  int Falt;
  
  // öppna passwd
  ifstream In ( "/etc/passwd" );
  
  // läs rader från filen
  while ( getline ( In, Rad ) ) {
	// nollställ alla index för varje ny rad
	Index1 = 0;
	Index2 = 0;
	Falt = 0;
	
	// iterera nu över varje rad och separera på ett ':'
	while ( (Index2 = Rad.find ( ':', Index1 )) != string::npos ) {
	  // har vi användarnamnet (fält 1)?
	  if ( Falt == 0 ) {
		cout.setf ( ios::left, ios::adjustfield );
		cout << setw(10) << Rad.substr ( Index1, Index2 - Index1 ).c_str() << " = ";
	  }
	  
	  // har vi fulla namnet (fält 5?
	  if ( Falt == 2 ) {
		cout.setf ( ios::right, ios::adjustfield );
		cout << setw(4) << Rad.substr ( Index1, Index2 - Index1 ).c_str() << endl;
	  }

	  // öka alla index
	  Index1 = Index2 + 1;
	  Falt++;
	}
  }
}
