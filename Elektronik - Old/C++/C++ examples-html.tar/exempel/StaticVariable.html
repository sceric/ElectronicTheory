#include  <stream.h>

void raknaAnrop () {
  // statisk variabel med värdet 0
  static int Anrop = 0;

  // öka antalet anrop med 1
  Anrop++;
  cout << "raknaAnrop har anroptats: " << Anrop << " gånger." << endl;
}

int main () {
  // anropa funktionen 'Anrop' några gånger
  for ( int Index = 0; Index < 5; Index++ ) {
    raknaAnrop ();
  } 
}
