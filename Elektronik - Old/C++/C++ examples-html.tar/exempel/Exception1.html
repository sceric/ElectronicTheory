//! program som visar hur en exception kan kastas och tas emot (throw/catch)
#include <iostream>
#include <string>

float divide (float X, float Y) {

  // är täljaren 0?
  if ( Y == 0 ) {
	// jep, kasta en exception
	throw string ("Nämnaren måste vara != 0");
  }
  
  // värdena ok, räkna och returnera
  return X / Y;
}

int main () {

  float X, Y;

  // evig slinga
  while ( 1 ) {
	cout << "Ge täljaren: ";
	cin >> X;
	cout << "Ge nämnaren: ";
	cin >> Y;

	// försök skriva ut kvoten och fånga fel
	try {
	  cout << X << "/" << Y << " = " << divide (X, Y) << endl;
	}
	catch (string Error) {
	  // ett fel har fångats, skriv ut meddelandet
	  cout << Error << endl;
	}
  }
}
