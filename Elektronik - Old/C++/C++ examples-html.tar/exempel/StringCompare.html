//! testar jämförelse av strängar
#include <iostream>
#include <string>

int main () {

  string S1 = "abc";
  string S2 = "ABC";
  string S3 = "Hej";
  
  if ( S1 < S1 )
	cout << S1 << " är mindre än " << S2 << endl;
  else
	cout << S1 << " är större än " << S2 << endl;

  if ( S2 < S3 )
	cout << S2 << " är mindre än " << S3 << endl;
  else
	cout << S2 << " är större än " << S3 << endl;

}
