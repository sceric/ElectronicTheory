//! demonstrerar hur SQL-satser kan skapas dynamiskt med ostrstream
#include <strstream>
#include <string>

int main () {
  int Salary;
  string Job;

  // läs in värden på lön och jobb
  cout << "Ge jobbtyp: ";
  cin >> Job;
  cout << "Ge lön:     ";
  cin >> Salary;
  
  // skapa SQL-satsen
  ostrstream SQL;

  SQL << "SELECT *" << "FROM PERSONS " << endl
      << "WHERE Job = '" << Job << "' AND Salary > "
      << Salary << ";";

  // skriv ut den nya satsen  
  cout << endl << "SQL-satsen är nu: " << endl
       << SQL.str () << endl;

}
