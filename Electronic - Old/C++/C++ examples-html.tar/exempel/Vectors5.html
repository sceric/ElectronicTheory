#include <iostream>
#include <iomanip>

int main () {
  int Tal [8][4];

  // fyll matrisen med data
  for ( int Y = 0; Y < 8; Y++ ) {
	for ( int X = 0; X < 4; X++ ) {
	  Tal [Y][X] = Y * 4 + X;
	}
  }

  // skriv ut matrisen till skärmen
  for ( int Y = 0; Y < 8; Y++ ) {
	for ( int X = 0; X < 4; X++ ) {
	  cout << setw (3) << Tal [Y][X] << " ";
	}
	cout << endl;
  }
}
