//! testar insättning i strängar
#include <iostream>
#include <string>

int main () {
  string Namn = "Frodo Bagger";

  // lägg till initialen för mellannamnet
  string Initial = "T. ";
  Namn.insert ( 6, Initial );

  // sätt in i början av strängen
  Namn.insert ( 0, "Mr. " );

  cout << "Namnet är: " << Namn << endl;
}
