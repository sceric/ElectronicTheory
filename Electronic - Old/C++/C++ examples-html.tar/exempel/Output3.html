#include <iostream>

int main () {

  double Tal = 1234567.89;

  // normal notation
  cout << Tal << endl;

  // alltid med decimalkomma
  cout.setf ( ios::fixed, ios::floatfield );
  cout << Tal << endl;

  // vetenskaplig notation
  cout.setf ( ios::scientific, ios::floatfield );
  cout << Tal <<endl;

  // normal notation
  cout.setf ( 0, ios::floatfield );
  cout << Tal <<endl;
}
