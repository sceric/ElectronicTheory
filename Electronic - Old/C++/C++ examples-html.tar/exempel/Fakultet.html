//! program som anropar en funktion rekursivt och beräknar n! (n-fakultet)
#include <iostream>

unsigned int Fakultet (unsigned int N) {
  if ( N == 0 ) {
    // basfallet nått
    return 1;
  }

  return N * Fakultet ( N - 1 );
}

int main () {
  unsigned int Tal;

  cout << "Ge in ett tal: ";
  cin >> Tal;

  // beräkna fakulteten och skriv ut svaret
  cout << Tal << "! = " << Fakultet ( Tal ) << endl;
}
