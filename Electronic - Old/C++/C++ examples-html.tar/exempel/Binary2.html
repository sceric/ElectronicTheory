//! läser in tal från en binär fil
#include <fstream>
#include <stdlib.h>

int main () {
  int Tmp;
  
  // öppna filen
  ifstream In ( "Test.bin");

  // kunde filen öppnas?
  if ( ! In ) {
	// kunde inte öppna filen!
	cout << "Kunde inte öppna filen 'Test.bin'!" << endl;
	exit ( EXIT_FAILURE );
  }

  // läs in tal från filen
  while ( In.read ( &Tmp, sizeof(int) ) ) {
	cout << "Läste: " << Tmp << endl;
  }
}
