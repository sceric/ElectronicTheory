//! program som demonstrerar en nästlad for-slinga
#include <iostream>

int main () {
  
  for ( int Rad = 0; Rad < 100; Rad += 10 ) {
	for ( int Kolumn = 0; Kolumn < 10; Kolumn++ ) {
	  // skriv ut ett tal 
	  cout << Rad + Kolumn << " ";
    }
    // en rad full, gå till nästa rad
    cout << "\n";
  }
}
