#include <iostream>

void in (int * Tal, int Antal) {
  // iterera och läs in tal
  for ( int Index = 0; Index < Antal; Index++ ) {
	cout << "Ge in tal " << Index + 1 << ": ";
	cin >> *(Tal + Index);
  }
}

void ut (int Tal[3]) {
  // skriv ut vårt tal
  for ( int Index = 0; Index < 3; Index++ ) {
	cout << "Tal " << Index << " är: " << Tal[Index] << endl;
  }
}

int main () {

  int Tal [3];

  // läs in och skriv sedan ut
  in ( Tal, 3 );
  ut ( Tal );

}
