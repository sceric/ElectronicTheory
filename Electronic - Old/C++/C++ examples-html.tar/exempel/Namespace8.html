//! visar hur aliaset till ett namespace kan implementeras

namespace DepartmentOfComputerScience {
  class Foo {
  public:
    Foo () {}
  };
}

int main () {
  // använd namespacet oförkortat
  DepartmentOfComputerScience::Foo F1;

  // definiera en alias
  namespace CS = DepartmentOfComputerScience;

  // använd aliasen
  CS::Foo F2;
}
