//! program som anropar en funktion rekursivt och beräknar ett Fibonacci-värde
#include  <iostream>

unsigned int fib (unsigned int N) {
  if ( N == 0 ) {
    // basfall 1 nått
    return 0;
  }

  if ( N == 1 ) {
    // basfall 2 nått
    return 1; 
  }

  // almänt fall, använd rekursion
  return fib ( N - 1 ) + fib ( N - 2 );
}

int main () {
  unsigned int Tal;
  cout << "Ge in ett tal: ";
  cin >> Tal;

  // beräkna fakulteten och skriv ut svaret
  cout << "fib(" << Tal << ") = " << fib ( Tal ) << endl;
}
