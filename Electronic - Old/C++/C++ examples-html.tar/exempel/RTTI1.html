#include <iostream>
#include <stdlib.h>

// definiera våra klasser
class A {
public:
  virtual void foo () { };
};
class B : public A { };
class C : public B { };

void doIt (A * Data) {
  C * ObjC;
  B * ObjB;

  // försök konvertera till ett 'C'
  if ( ( ObjC = dynamic_cast<C *>(Data) ) != 0 ) {
    cout << "Objektet av typen C" << endl;
  }

  // ett 'B'?
  else if ( ( ObjB = dynamic_cast<B *>(Data) ) != 0 ) {
    cout << "Objektet av typen B" << endl;
  }

  // måste vara ett 'A'
  else {
    cout << "Objektet av typen A" << endl;
  }
}


int main () {

  A * Tmp;
  
  // iterera och skapa 10 slumpmässiga objekt
  for ( int Index = 0; Index < 10; Index++ ) {
    int Type = rand () % 3;

    cout << "Skapade ett " << (char)('A' + Type) << endl;

    // vad blev det?
    switch ( Type ) {
    case 0 :
      Tmp = new A;
      doIt ( Tmp );
      break;
    case 1 :
      Tmp = new B;
      doIt ( Tmp );
      break;
    case 2 :
      Tmp = new C;
      doIt ( Tmp );
      break;
    }

    // frigör minne
    delete Tmp;
  }
}
