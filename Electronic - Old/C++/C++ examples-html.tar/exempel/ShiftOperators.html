#include <stream.h>

int main () {

   int Tal = 43;
   
   cout << Tal << " << 3 = " << (Tal << 3) << endl;
   cout << Tal << " << 2 = " << (Tal << 2) << endl;
   cout << Tal << " << 1 = " << (Tal << 1) << endl;
   cout << Tal << " >> 2 = " << (Tal >> 2) << endl;
   cout << Tal << " >> 1 = " << (Tal >> 1) << endl;
}
