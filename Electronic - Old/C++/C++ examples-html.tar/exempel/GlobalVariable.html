//! demonstrerar hur en global variabel kan deklareras och användas
#include <iostream>

// global variabel
int Global = 10;

void testaGlobal () {
  // sätt ett nytt värde på den globala variabeln
  Global = 20;
}

int main () {
  cout << "Global har värdet: " << Global << endl;
  testaGlobal ();
  cout << "Global har värdet: " << Global << endl;
}
