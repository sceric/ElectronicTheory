//! öppnar och skriver till en fil
#include <fstream>
#include <stdlib.h>

int main () {
  // öppna filen
  ofstream Ut ( "index.html" );

  // kunde filen öppnas?
  if ( ! Ut ) {
    // kunde inte öppna filen!
    cout << "Kunde inte öppna filen 'index.html'!" << endl;
    exit ( EXIT_FAILURE );
  }

  // skriv en html-version av 'Hello world'
  Ut << "<html><head>" << endl;
  Ut << "<title>Hello world</title>" << endl;
  Ut << "<body>" << endl;
  Ut << "Hello world" << endl;
  Ut << "</body></html>" << endl;
}
