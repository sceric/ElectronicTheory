//! konkatenerar en C-strängar till en annan
#include <iostream>
#include <string.h>
  
int main () {
  char Destination[100] = "En exempelsträng";
  char Source [] = ", och en annan";

  // utför kopieringen 
  strcat ( Destination, Source ); 

  cout << "Source      = '" << Source<< "', " << endl
       << "Destination = '" << Destination << "'" << endl;
}
