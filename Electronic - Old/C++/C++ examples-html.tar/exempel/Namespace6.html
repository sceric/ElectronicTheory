//! visar hur man kan använda endast vissa symboler ur ett namespace
#include <iostream>
#include <string>

int main () {
  // använd string från namespacet std
  using std::string;

  string Ord;
  
  // läs in ett år
  std::cout << "Skriv ett ord: ";
  std::cin >> Ord;

  // skriv vår hälsning
  std::cout << "Du skrev in: " << Ord << endl;
}
