//! använder en iterator för att skriva ut tecknen i en sträng

#include <iostream>
#include <string>

int main () {

   string S = "Hello";

   // iterator till strängens tecken
   string::iterator It = S.begin ();

   // skriv ut alla tecken genom att iterera över dem
   for ( ; It != S.end (); It++ ) {
	 cout << *It << endl;
   }
}
