//! demonstrerar hur enskilda ord kan läsas ur en sträng med istrstream
#include <strstream>
#include <string>

int main () {
  string Rad;
  string Ord;
  int Antal = 1;
  
  // läs in en rad
  cout << "Ge en rad med text: ";
  getline ( cin, Rad );
  
  // skapa stream
  istrstream Stream ( Rad.c_str () );

  // iterera så länge det finns flera ord
  while ( Stream >> Ord ) {
    cout << "Ord " << Antal++ << " = " << Ord << endl;
  }
}
