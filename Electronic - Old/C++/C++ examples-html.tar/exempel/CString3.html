//! demonsterar felaktig tilldelning av C-strängar
#include <iostream>

int main () {
  char Hello1[] = "Hello 1!";
  char * Hello2;

  // skriv ut Hello1
  cout << "Hello1 = '" << Hello1 << "'" << endl << endl;
  
  // tilldelning
  Hello2 = Hello1;

  // manipulera Hello2
  Hello2 [6] = '2';
  
  // skriv ut Hello1 och Hello2
  cout << "Hello1 = '" << Hello1 << "'" << endl;
  cout << "Hello2 = '" << Hello2 << "'" << endl;
}
