//! kopierar en C-sträng med en funktion copy(). Allokerar minne.
#include <iostream>
#include <string.h>

int main () {
  char Hello1[] = "Hej! Detta är en lååång sträng.";
  char * Hello2;

  // allokera minne
  Hello2 = new char [ strlen ( Hello1 ) + 1 ];
  
  // utför kopieringen 
  strcpy ( Hello2, Hello1 );

  cout << "Hello1 = '" << Hello1 << "', " << endl
       << "Hello2 = '" << Hello2 << "'" << endl;

  delete [] Hello2;
}
