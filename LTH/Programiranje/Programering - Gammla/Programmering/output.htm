Utskrifter med Writer-klasserna

Följande program illustrerar hur man kan göra för att formatera utskrifter med hjälp av Writer-klasserna i se.lth.cs.pt.io-paketet.

Lägg märke till att det bara är print-operationen som kan använda formateringssträngar, men att man genom att infoga \n får en radbrytning (så println-operationen är egentligen onödig). 

    import se.lth.cs.pt.io.*;
    import se.lth.cs.pt.random.*;
    
    public class Output {
    
    	public static void main(String[] args) {
    	    Writer writer = new ConsoleWriter();
    
    	    double x = Math.PI;
    	    writer.print("x = %.5f\n", x);
    	    writer.print("x = %5.2f\n", x);
    
    	    double[] v = {4.25, 14.154, 133,1, 0.4, -2.4};
    	    writer.println("Högerjusterat:");
    	    for (int k = 0; k < v.length; k++) {
    		writer.print(" v[%i] = ", k);
    		writer.print("%6.2f\n", v[k]);
    	    }
    	    writer.println("Vänsterjusterat:");
    	    for (int k = 0; k < v.length; k++) {
    		writer.print(" v[%i] = ", k);
    		writer.print("%-6.2f\n", v[k]);
    	    }
    	    writer.println("Vänsterjusterat, med tecken utskrivet:");
    	    for (int k = 0; k < v.length; k++) {
    		writer.print(" v[%i] = ", k);
    		writer.print("%+6.2f\n", v[k]);
    	    }
    
    	    v = new double[15];
    	    RandomNumberGenerator rng = new RandomNumberGenerator();
    	    for (int k = 0; k < v.length; k++) {
    		v[k] = rng.randDouble(-100,100);
    	    }
    	    writer.println("Högerjusterat index och högerjusterade värden:");
    	    for (int k = 0; k < v.length; k++) {
    		writer.print(" v[%2i] = ", k);
    		writer.print("%6.2f\n", v[k]);
    	    }
    	}
    }
