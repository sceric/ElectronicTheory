class Card {

    private int color, value;

    public Card(int color, int value) {
        this.color = color;
        this.value = value;
    }

    public int getColor() {
        return this.color;
    }

    public int getValue() {
        return this.value;
    }
}

class CardDeck {

    private Card[] deck;
    private int nbrOfCards;

    public CardDeck() {
        this.nbrOfCards = 0;
        this.deck = new Card[52];
        for (int color = 1; color <= 4; color++) {
            for (int value = 1; value <= 13; value++) {
                this.deck[this.nbrOfCards++] = new Card(color,value);
            }
        }
    }

    public void shuffle() {
        RandomNumberGenerator rng = new RandomNumberGenerator();
        for (int k = 51; k >= 1; k--) {
            int s = rng.randInt(0,k);
            Card temp = this.deck[s];
            this.deck[s] = this.deck[k];
            this.deck[k] = temp;
        }
    }

    public boolean cardsLeft() {
        return this.nbrOfCards > 0;
    }

    public Card getCard() {
        return this.deck[--this.nbrOfCards];
    }
}
