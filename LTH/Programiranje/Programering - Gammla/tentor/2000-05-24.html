    class Control {
    
        private double distance;
        private SplitTime[] splitVec;
        private int nbrPassed;
    
        public Control(double dist, int maxNbrPassed) {
            this.distance = dist;
            this.splitVec = new SplitTime[maxNbrPassed];
            this.nbrPassed = 0;
        }
    
        public double getDistance() {
            return this.distance;
        }
    
        public void runnerPassed(Runner runner, int clock) {
            runner.setLastControl(this);
            SplitTime splitTime = new SplitTime(runner, clock - runner.getStartTime());
            int pos = this.nbrPassed;
            while (pos > 0 && splitTime.getCTime() < this.splitVec[pos-1].getCTime()) {
                this.splitVec[pos] = this.splitVec[pos-1];
                pos--;
            }
            this.splitVec[pos] = splitTime;
            this.nbrPassed++;
        }
    
        public int getNbrPassed() {
            return this.nbrPassed;
        }
    
        public SplitTime getSplitTime(int place) {
            return this.splitVec[place-1];
        }
    
        public int getClock(Runner runner) {
            int pos = 0;
            while (pos < this.nbrPassed && this.splitVec[pos].getRunner() != runner) {
                pos++;
            }
            if (pos < this.nbrPassed) {
                return runner.getStartTime() + this.splitVec[pos].getCTime();
            } else {
                return -1;
            }
        }

                writer.println(i + ". " +
                               splitTime.getRunner().getNbr() + ", " +

                               hours + ":" + minutes + "." + seconds);
            }
        }
    
        public void reportRunner(int nbr, Writer writer) {
            Runner runner = this.runnerVec[nbr];
            Control lastControl = runner.getLastControl();
            if (lastControl == null) {
                writer.println(nbr + ", " +
                               runner.getName() +
                               " har inte passerat någon kontroll.");
            } else {
                int time = runner.getStartTime() + lastControl.getClock(runner),
                    hour = time / 3600,
                    minute = (time / 60) % 60;
                writer.println(nbr + ", " +
                               runner.getName() +
                               " passerade kontrollen " +
                               lastControl.getDistance() +
                               " klockan " +
                               hour + "." + minute);
            }
        }
    }

    class Control {
    
        private double distance;
        private Head list;
    
        public Control(double dist) {
            this.distance = dist;
            this.list = new Head();
        }
    
        public void runnerPassed(Runner runner, int clock) {
            runner.setLastControl(this);
            SplitTime
                splitTime = new SplitTime(runner, clock - runner.getStartTime()),
                e = (SplitTime) this.list.first();
            while (e != null && e.getCTime() < splitTime.getCTime()) {
                e = (SplitTime) e.suc();
            }
            if (e == null) {
                splitTime.into(this.list);
            } else {
                splitTime.precede(e);
            }
        }
    
        public int getNbrPassed() {
            return this.list.cardinal();
        }
    
        public SplitTime getSplitTime(int place) {
            SplitTime e = (SplitTime) this.list.first();
            for (int i = 2; i <= place; i++) {
                e = (SplitTime) e.suc();
            }
            return e;
        }
    
        public int getClock(Runner runner) {
            SplitTime e = (SplitTime) this.list.first();
            while (e != null && e.getRunner() != runner) {
                e = (SplitTime) e.suc();
            }
            if (e == null) {
                return -1;
            } else {
                return runner.getStartTime() + e.getCTime();
            }
        }
    }
